\section*{Notation}

\begin{align*}
B &= \text{number of root states searched in parallel},\\
A &= \text{maximum number of actions},\\
K &= \text{number of Thompson samples per root state per round},\\
R &= \text{number of posterior update rounds},\\
H &= \text{internal rollout horizon after the root action},\\
M &= \text{number of Monte Carlo samples for the policy target},\\
s &\in \mathcal{S}^{B},\\
\alpha &\in \R_{>0}^{B \times A \times 3},\\
a &\in \N^{B \times K},\\
d &\in \Delta^2,\\
\lambda &> 0.
\end{align*}

The utility used for search selection and policy improvement is
\[
U(\phi) = \phi_W - \phi_L.
\]

\section*{Generic Batched Thompson Root Search}

\begin{algorithmblock}{Algorithm 1: Dirichlet Q-head parameters}
\textbf{Input:} state $s$, action $a$, network parameters $\theta$\\
$\bar{\phi}_\theta^Q(s,a) \gets \softmax(r_\theta^Q(s,a))$\\
$\alpha_{\theta,0}^Q(s,a) \gets \softplus(t_\theta^Q(s,a))$\\
$\alpha_\theta^Q(s,a) \gets \alpha_{\theta,0}^Q(s,a)\bar{\phi}_\theta^Q(s,a)$\\
\textbf{return} $\alpha_\theta^Q(s,a)$
\end{algorithmblock}

\begin{algorithmblock}{Algorithm 2: ThompsonSampleActions$(\alpha,K,\legal)$}
\textbf{Input:} $\alpha \in \R_{>0}^{B \times A \times 3}$, $K$, $\legal \in \{0,1\}^{B \times A}$\\
Sample $\phi \sim \Dirichlet(\alpha)$ with $\phi \in \R^{B \times K \times A \times 3}$\\
$u \gets U(\phi)$\\
$u \gets \MaskIllegal(u,\legal,-\infty)$\\
$a \gets \arg\max_{\mathrm{action}} u$\\
\textbf{return} $a$
\end{algorithmblock}

\begin{algorithmblock}{Algorithm 3: GetEvidence$(s_f,\theta)$}
\textbf{Input:} final rollout state $s_f$, network parameters $\theta$\\
\textbf{if} $s_f$ is terminal with outcome $z$ \textbf{then}\\
\> $d \gets e_z$\\
\> $\lambda \gets c_{\mathrm{terminal}}$\\
\textbf{else}\\
\> $d \gets \softmax(r_\theta^V(s_f))$\\
\> $\lambda \gets c_{\mathrm{leaf}}$\\
\textbf{end if}\\
$d \gets \fliproot(d)$\\
\textbf{return} $(d,\lambda)$
\end{algorithmblock}

\begin{algorithmblock}{Algorithm 4: InternalRollout$(s,a,\theta,H)$}
\textbf{Input:} root state batch $s$, root actions $a$, network parameters $\theta$, horizon $H$\\
$s' \gets \Step(s,a)$\\
\textbf{for} $h=1,\ldots,H$ \textbf{do}\\
\> $\pi \gets \softmax(\ell_\theta(s',\cdot))$\\
\> $a' \gets \PolicyAction(\pi)$\\
\> $s' \gets \Step(s',a')$\\
\textbf{end for}\\
$(d,\lambda) \gets \operatorname{GetEvidence}(s',\theta)$\\
\textbf{return} $(d,\lambda)$
\end{algorithmblock}

\begin{algorithmblock}{Algorithm 5: UpdateAlphas$(\alpha,a,d,\lambda)$}
\textbf{Input:} current posterior $\alpha$, sampled actions $a$, rollout evidence $(d,\lambda)$\\
$\Delta\alpha_{b,k} \gets \lambda_{b,k}d_{b,k}$\\
\textbf{for each} batch element $b$ and action $j$ \textbf{do}\\
\> $\alpha_{b,j}^{\mathrm{new}}
   \gets
   \alpha_{b,j}
   +
   \sum_{k:a_{b,k}=j}\Delta\alpha_{b,k}$\\
\textbf{end for}\\
\textbf{return} $\alpha^{\mathrm{new}}$
\end{algorithmblock}

\begin{algorithmblock}{Algorithm 6: BatchedThompsonRootSearch$(s,\theta,R,K,H)$}
\textbf{Input:} root state batch $s$, network parameters $\theta$, rounds $R$, samples $K$, horizon $H$\\
$\alpha \gets \alpha_\theta^Q(s,\cdot)$\\
$\legal \gets \LegalActions(s)$\\
\textbf{for} $r=1,\ldots,R$ \textbf{do}\\
\> $a \gets \operatorname{ThompsonSampleActions}(\alpha,K,\legal)$\\
\> $(d,\lambda) \gets \operatorname{InternalRollout}(s,a,\theta,H)$\\
\> $\alpha \gets \operatorname{UpdateAlphas}(\alpha,a,d,\lambda)$\\
\textbf{end for}\\
$\alpha_{\mathrm{search}}(s,\cdot) \gets \alpha$\\
\textbf{return} $\alpha_{\mathrm{search}}(s,\cdot)$
\end{algorithmblock}

\section*{Search-Improved Policy Target}

\begin{algorithmblock}{Algorithm 7: PosteriorBestPolicyTarget$(\alpha_{\mathrm{search}},M,\legal)$}
\textbf{Input:} search posterior $\alpha_{\mathrm{search}} \in \R_{>0}^{B \times A \times 3}$, samples $M$, legal mask $\legal$\\
Sample $\phi^{(m)}_b \sim \Dirichlet(\alpha_{\mathrm{search}}(b,\cdot))$ for $m=1,\ldots,M$\\
$u^{(m)}_{b,a} \gets U(\phi^{(m)}_{b,a})$\\
$u^{(m)}_{b,a} \gets \MaskIllegal(u^{(m)}_{b,a},\legal,-\infty)$\\
$a^\star_{b,m} \gets \arg\max_a u^{(m)}_{b,a}$\\
$\hat{\pi}_{\mathrm{search}}(a \mid s_b)
  \gets
  \frac{1}{M}\sum_{m=1}^{M}\mathbf{1}[a^\star_{b,m}=a]$\\
\textbf{return} $\hat{\pi}_{\mathrm{search}}(\cdot \mid s)$
\end{algorithmblock}

Equivalently,
\[
\pi_{\mathrm{search}}(a \mid s)
=
\mathbb{P}
\left(
a =
\arg\max_b U(\phi_b)
\right),
\qquad
\phi_b
\sim
\Dirichlet\left(\alpha_{\mathrm{search}}(s,b)\right).
\]

\section*{MCTX Implementation of Algorithm 6 Used by \texttt{play.py}}

Algorithms 8 and 9 are not a second search rule. They describe how
\texttt{play.py} realizes Algorithm 6 inside the fixed-size MCTX tree
machinery. MCTX owns the simulation loop and tree storage. The Dirichlet-Q
logic owns the root action selection rule and the reconstruction of the live
root posterior from evidence stored in the tree.

For a search budget $T$, MCTX allocates
\[
N=T+1
\]
nodes per root state. Node $0$ is the root, and each simulation can expand at
most one additional node. The tree contains array fields such as
\[
\begin{aligned}
\mathrm{node\_visits} &\in \N^{B \times N},\\
\mathrm{parents} &\in \N^{B \times N},\\
\mathrm{action\_from\_parent} &\in \N^{B \times N},\\
\mathrm{children\_index} &\in \N^{B \times N \times A},\\
\mathrm{children\_prior\_logits} &\in \R^{B \times N \times A},\\
\mathrm{children\_visits} &\in \N^{B \times N \times A}.
\end{aligned}
\]

In this implementation, the important custom field is the per-node embedding.
For each tree node $n$, \texttt{play.py} stores
\[
\mathrm{embedding}_n =
\left(
s_n,\,
d_n,\,
w_n,\,
a^{\mathrm{root}}_n,\,
p_n,\,
\alpha_{\theta}^Q(s_n,\cdot)
\right),
\]
where $s_n$ is the environment state at the node, $d_n$ is the WDL
distribution in the node-local player-to-move perspective, $w_n$ is the
Dirichlet evidence weight, $a^{\mathrm{root}}_n$ is the first action taken from
the root on the path to $n$, and $p_n \in \{0,1\}$ is the depth parity relative
to the root.

When MCTX expands a child by action $a$ from a parent node, \texttt{play.py}
applies the environment step and evaluates the resulting observation:
\[
(s', \ell_\theta(s',\cdot), \alpha_\theta^V(s'), \alpha_\theta^Q(s',\cdot)).
\]
If $s'$ is terminal, the stored WDL distribution is terminal one-hot evidence
in the child-local perspective and $w_n=c_{\mathrm{terminal}}$. Otherwise, the
stored WDL distribution is the value-head mean
\[
d_n = \bar{\phi}_\theta^V(s')
\]
and $w_n=c_{\mathrm{leaf}}$. The root action is set to the action taken from
the root at depth one and then inherited by all deeper descendants. The depth
parity flips at every ply.

The accumulated root evidence represented by the tree is
\[
\RootEvidenceSum(\mathrm{tree})_{b,a}
=
\sum_{n=1}^{N-1}
\mathbf{1}
\left[
a^{\mathrm{root}}_{b,n}=a
\;\wedge\;
\mathrm{node\_visits}_{b,n}>0
\right]
w_{b,n}\,
\alignroot(d_{b,n},p_{b,n}),
\]
where $\alignroot(d,p)$ flips the WDL vector when $p=1$ and leaves it unchanged
when $p=0$. Thus all evidence is accumulated in the root player's perspective
and bucketed by the first action chosen from the root.

Each MCTX simulation proceeds as follows:
\[
\begin{aligned}
&\text{select a path from the root,}\\
&\text{expand one edge with the recurrent function,}\\
&\text{store node evidence,}\\
&\text{run MCTX's visit/value backup.}
\end{aligned}
\]
The three MCTX subroutines in Algorithm 9 have the following roles.

\begin{itemize}
\item
\textbf{MCTXSimulate} starts at the root node. At depth $0$ it calls
\texttt{MCTXRootThompsonSelect}. At deeper nodes it calls the configured
interior selector. It follows already-expanded child pointers
$\mathrm{children\_index}$ until it finds an unvisited edge, or until the
maximum depth is reached. It returns the parent node index $i$ and action $a$
to evaluate next. This step only reads the tree; it does not call the
environment or the network.

\item
\textbf{MCTXExpandAndStoreEvidence} evaluates the selected edge $(i,a)$. It
reads the parent embedding, applies $\Step(s_i,a)$, evaluates the resulting
state with the network, and writes a child node into the tree. The stored child
embedding contains the new environment state, the WDL evidence $d_n$, the
evidence weight $w_n$, the inherited root action $a_n^{\mathrm{root}}$, the
flipped depth parity $p_n$, and the local Q prior
$\alpha_\theta^Q(s_n,\cdot)$. This step also records the tree topology:
parent pointer, action-from-parent, child index, prior logits, reward, and
discount.

\item
\textbf{MCTXBackward} runs MCTX's ordinary scalar backup from the expanded
child back to the root. Along the path it increments node and edge visit
counts and updates averaged scalar values using the recurrent reward and
discount. These scalar values are useful for MCTX bookkeeping and interior
selection, but the Dirichlet-Q root posterior is reconstructed from the stored
WDL evidence through $\RootEvidenceSum(\mathrm{tree})$.
\end{itemize}

The final policy target does not use MCTX visit counts directly. It uses the
Dirichlet posterior
\[
\alpha_{\mathrm{search}}
=
\alpha_\theta^Q(s,\cdot)
+
\RootEvidenceSum(\mathrm{tree})
\]
and then applies Algorithm 7.
Equivalently, after simulation $t$ inside one MCTX block, the live root
posterior is
$\alpha_{\mathrm{block}}^{(t)}
=\alpha_{\mathrm{base}}+\RootEvidenceSum(\mathrm{tree})$.
The implementation does not store $\alpha^{(t)}$ as a separate mutable array;
it reconstructs it from the current tree whenever root Thompson selection is
called.
Within one call to Algorithm 9,
\[
K=1.
\]
Each pass through the loop is one Thompson root selection. The call to
\texttt{MCTXSimulate} begins at the root, so it calls
\texttt{MCTXRootThompsonSelect} once per pass, using the posterior
$\alpha^{(t-1)}$ accumulated from all previous expanded evidence.

\begin{algorithmblock}{Algorithm 8: MCTXRootThompsonSelect$(\mathrm{tree})$}
$\alpha_{\mathrm{base}} \gets \mathrm{tree.extra\_data.alpha\_Q\_prior}$\\
$E \gets \RootEvidenceSum(\mathrm{tree})$\\
$\alpha_{\mathrm{post}} \gets \alpha_{\mathrm{base}} + E$\\
Sample $\phi \sim \Dirichlet(\alpha_{\mathrm{post}})$\\
$u \gets U(\phi)$\\
$u \gets \MaskIllegal(u,\mathrm{tree.root\_invalid\_actions},-\infty)$\\
$a \gets \arg\max_{\mathrm{action}} u$\\
\textbf{return} $a$
\end{algorithmblock}

\begin{algorithmblock}{Algorithm 9: MCTXDirichletRootSearch$(s,\theta,T,\alpha_{\mathrm{base}})$}
\textbf{Input:} root state $s$, network parameters $\theta$, simulations $T$, starting posterior $\alpha_{\mathrm{base}}$\\
\textbf{if} $\alpha_{\mathrm{base}}$ is omitted \textbf{then} $\alpha_{\mathrm{base}} \gets \alpha_\theta^Q(s,\cdot)$ \textbf{end if}\\
$\mathrm{tree} \gets \operatorname{InstantiateTreeFromRoot}(s,\theta,N=T+1,\alpha_{\mathrm{base}})$\\
$E_0 \gets 0$\\
$\alpha_{\mathrm{block}}^{(0)} \gets \alpha_{\mathrm{base}}$\\
\textbf{for} $t=1,\ldots,T$ \textbf{do}\\
\> \textit{Root Thompson selection uses $\alpha_{\mathrm{block}}^{(t-1)}$}\\
\> $(i,a) \gets
   \operatorname{MCTXSimulate}
   (\mathrm{tree},
    \operatorname{root\_selection}=\operatorname{MCTXRootThompsonSelect})$\\
\> $\mathrm{tree} \gets
   \operatorname{MCTXExpandAndStoreEvidence}(\mathrm{tree},i,a,\theta)$\\
\> $\mathrm{tree} \gets \operatorname{MCTXBackward}(\mathrm{tree})$\\
\> $E_t \gets \RootEvidenceSum(\mathrm{tree})$\\
\> $\alpha_{\mathrm{block}}^{(t)} \gets \alpha_{\mathrm{base}} + E_t$\\
\textbf{end for}\\
$E_T \gets \RootEvidenceSum(\mathrm{tree})$\\
$\alpha_{\mathrm{search}} \gets \alpha_{\mathrm{block}}^{(T)} = \alpha_{\mathrm{base}} + E_T$\\
\textbf{return} $(\mathrm{tree},E_T,\alpha_{\mathrm{search}})$
\end{algorithmblock}

\begin{algorithmblock}{Algorithm 10: RepeatedMCTXDirichletRootSearch$(s,\theta,G,T)$}
\textbf{Input:} root state $s$, network parameters $\theta$, outer rounds $G$, simulations per round $T$\\
$\alpha^{(0)} \gets \alpha_\theta^Q(s,\cdot)$\\
$E_{\mathrm{total}} \gets 0$\\
\textbf{for} $g=1,\ldots,G$ \textbf{do}\\
\> $(\mathrm{tree}_g,E_g,\alpha^{(g)})
   \gets
   \operatorname{MCTXDirichletRootSearch}(s,\theta,T,\alpha^{(g-1)})$\\
\> $E_{\mathrm{total}} \gets E_{\mathrm{total}} + E_g$\\
\textbf{end for}\\
$\alpha_{\mathrm{search}} \gets \alpha^{(G)}
  = \alpha_\theta^Q(s,\cdot) + E_{\mathrm{total}}$\\
\textbf{return} $(\{\mathrm{tree}_g\}_{g=1}^{G},E_{\mathrm{total}},\alpha_{\mathrm{search}})$
\end{algorithmblock}

For the planned implementation,
\[
T = \texttt{num\_simulations},
\qquad
M = \texttt{policy\_mc\_samples},
\]
and the current \texttt{play.py} search shape is closest to $G=1$. Algorithm
10 is the planned outer wrapper for repeated MCTX blocks; it performs $G$ calls
to Algorithm 9 and carries the posterior $\alpha^{(g)}$ into the next call.
